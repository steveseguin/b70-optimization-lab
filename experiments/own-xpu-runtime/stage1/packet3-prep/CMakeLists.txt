cmake_minimum_required(VERSION 3.25)
# Pin the compiler observed in ../../data/toolchain-20261010.json. Never run a
# SYCL try_run: CMake's compiler checks create static archives only.
set(CMAKE_CXX_COMPILER "/opt/intel/oneapi/compiler/2026.0/bin/icpx" CACHE FILEPATH "Pinned icpx")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(own_rt_packet3 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "2026.0.0")
  message(FATAL_ERROR "Requires the recorded IntelLLVM/icpx 2026.0.0 toolchain")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "MockDevice only: no SYCL headers, flags or libraries" ON)
option(OWN_RT_BUILD_NATIVE_SMOKE "Link native smoke (execution separately authorized)" OFF)
add_library(owner STATIC src/owner.cpp)
target_include_directories(owner PUBLIC src)
target_compile_options(owner PRIVATE -Wall -Wextra -Werror)
if(OWN_RT_HOST_ONLY)
  if(OWN_RT_BUILD_NATIVE_SMOKE)
    message(FATAL_ERROR "Native smoke cannot be enabled in host-only mode")
  endif()
  add_executable(own-rt-smoke src/smoke.cpp)
  target_compile_definitions(own-rt-smoke PRIVATE OWN_RT_HOST_ONLY=1)
  target_link_libraries(own-rt-smoke PRIVATE owner)
  add_executable(owner-tests tests/owner_tests.cpp)
  target_link_libraries(owner-tests PRIVATE owner)
  enable_testing()
  foreach(case census lifetime closed inflight order copy_bounds dependencies graph_lifetime marker receipt)
    add_test(NAME ${case} COMMAND owner-tests ${case})
  endforeach()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_test(NAME receipt_schema COMMAND ${Python3_EXECUTABLE} -B ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_receipt.py $<TARGET_FILE:owner-tests>)
  add_test(NAME mock_smoke COMMAND own-rt-smoke --enumerate --card 0 --receipt ${CMAKE_BINARY_DIR}/mock-smoke.json)
else()
  execute_process(COMMAND dpkg-query -W [=[-f=${Version}]=] libze-dev OUTPUT_VARIABLE ZE_VERSION RESULT_VARIABLE ZE_RESULT)
  if(NOT ZE_RESULT EQUAL 0 OR NOT ZE_VERSION MATCHES "^1\\.28\\.2")
    message(FATAL_ERROR "Requires recorded libze-dev 1.28.2 headers (API version is separately numbered)")
  endif()
  find_path(ZE_INCLUDE level_zero/ze_api.h PATHS /usr/include NO_DEFAULT_PATH REQUIRED)
  # OBJECT targets compile device code without producing a native executable.
  target_compile_options(owner PRIVATE -fsycl)
  add_library(sycl-compile OBJECT src/sycl_device.cpp src/smoke.cpp)
  target_include_directories(sycl-compile PRIVATE src ${ZE_INCLUDE})
  target_compile_options(sycl-compile PRIVATE -fsycl -Wall -Wextra -Werror)
  if(OWN_RT_BUILD_NATIVE_SMOKE)
    add_executable(own-rt-smoke $<TARGET_OBJECTS:sycl-compile>)
    target_link_libraries(own-rt-smoke PRIVATE owner ze_loader)
    target_link_options(own-rt-smoke PRIVATE -fsycl)
  endif()
endif()
