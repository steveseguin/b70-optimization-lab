cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet51 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "CPU mock only" ON)
set(P ${CMAKE_CURRENT_SOURCE_DIR})
set(R ${P}/../packet45-merged3)
if(NOT DEFINED P51_BASELINE_MANIFEST)
 message(FATAL_ERROR "Use build_layer.py: baseline inputs must be hash validated")
endif()
include(${P51_BASELINE_MANIFEST})
find_package(OpenSSL REQUIRED)
add_executable(layer-bench ${P51_SOURCE_MIRROR}/layer_bench.cpp ${P51_BASELINE_OBJECTS})
target_include_directories(layer-bench PRIVATE ${R} ${P}/../../stage1/packet3-prep/src ${P}/../packet2/kernels ${P}/../packet2/loader)
target_compile_options(layer-bench PRIVATE -O2 -fno-fast-math -ffp-contract=off -Werror=return-type)
target_link_options(layer-bench PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(layer-bench PRIVATE OpenSSL::Crypto)
if(OWN_RT_HOST_ONLY)
 add_executable(candidate-cpu-test ${P51_SOURCE_MIRROR}/candidate_cpu_test.cpp ${P51_BASELINE_OBJECTS})
 target_include_directories(candidate-cpu-test PRIVATE ${R} ${P}/../../stage1/packet3-prep/src ${P}/../packet2/kernels ${P}/../packet2/loader)
 target_compile_definitions(candidate-cpu-test PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(candidate-cpu-test PRIVATE -O2 -fno-fast-math -ffp-contract=off -fopenmp -Werror=return-type)
 target_link_options(candidate-cpu-test PRIVATE -fno-fast-math -ffp-contract=off -fopenmp)
 target_link_libraries(candidate-cpu-test PRIVATE OpenSSL::Crypto)
 target_compile_definitions(layer-bench PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(layer-bench PRIVATE -fopenmp)
 target_link_options(layer-bench PRIVATE -fopenmp)

else()
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Device build needs IntelLLVM")
 endif()
 target_compile_options(layer-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_options(layer-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_libraries(layer-bench PRIVATE ze_loader)
 get_filename_component(COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
 set_property(TARGET layer-bench PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
endif()
