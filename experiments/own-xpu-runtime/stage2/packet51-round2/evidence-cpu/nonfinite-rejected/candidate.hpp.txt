// IQ2_S has three magnitudes. An index selects sixteen bits rather than two
// 32-bit table entries. This is a codebook representation, not a weight repack.
consteval auto compact_iq2_table(){
 std::array<uint16_t,1024> out{};
 for(unsigned i=0;i<1024;++i)for(unsigned j=0;j<8;++j){
  unsigned v=p28::iq2_bytes[i*8+j];
  if(v!=8&&v!=25&&v!=43)throw "unexpected IQ2 magnitude";
  out[i]|=uint16_t((v==8?0:v==25?1:2)<<(j*2));
 }
 return out;
}
inline constexpr auto iq2_codes=compact_iq2_table();
template<unsigned K> inline void iq2_decode8(const uint8_t* row,unsigned k,float (&v)[8]){
 constexpr auto layout=p48::layout(Type::IQ2_S,K);
 unsigned block=k/256,i=k%256,q=i/8;
 unsigned pair=p48::load16(row+block*64+q*2);
 unsigned index=(pair&255)|(((row[layout.high+block*8+q/4]>>(2*(q%4)))&3)<<8);
 unsigned codes=iq2_codes[index],signs=pair>>8;
 float d=iq3::half(p48::load16(row+layout.halves+block*2));
 float scale=(d*(0.5f+float((row[layout.scales+block*8+i/32]>>(4*((i/16)%2)))&15)))*0.25f;
 bool exceptional=(std::bit_cast<uint32_t>(d)&0x7fffffff)>=0x7f800000||(std::bit_cast<uint32_t>(scale)&0x7fffffff)>0x7f800000;
 #pragma unroll
 for(unsigned j=0;j<8;++j){unsigned code=(codes>>(j*2))&3;int mag=code==0?8:code==1?25:43;
  if(exceptional)v[j]=(scale*float(mag))*float((signs>>j)&1?-1:1);
  else v[j]=scale*float((signs>>j)&1?-mag:mag);
 }
}
template<unsigned K> inline float gate_lane(const uint8_t* w,const uint16_t* x,unsigned lane){

   uint16_t bits=input[j];

   uint16_t bits=x[k+j];
#endif
   float product=value[j]*iq3::bits_float(uint32_t(bits)<<16);sum=sum+product;
  }
 }

}
