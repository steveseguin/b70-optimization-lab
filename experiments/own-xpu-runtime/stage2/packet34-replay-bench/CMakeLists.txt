cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet34 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(HOST_ONLY "CPU only" ON)
option(SANITIZE "ASan/UBSan" OFF)
set(P ${CMAKE_CURRENT_SOURCE_DIR})
set(R ${P}/../packet29-merged)
set(O ${P}/../../stage1/packet3-prep/src)
set(K ${P}/../packet2)
find_package(OpenSSL REQUIRED)
set(PRECISE -fno-fast-math -ffp-contract=off)
set(SOURCES p28/submit.cpp expert_down.cpp round2_submit.cpp small_gemv.cpp p25_bridge.cpp performance.cpp adapter.cpp specialised_submit.cpp runtime_kernels.cpp owner.cpp owner_retire.cpp ../packet2/loader/gguf.cpp)
if(NOT HOST_ONLY)
 list(APPEND PRECISE -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 list(APPEND SOURCES native.cpp ../packet2/kernels/device.cpp ../packet6-kopt/kernels.cpp ../packet7-kopt2/kernels.cpp ../packet8-kopt3/kernels.cpp ../packet9-kopt4/kernels.cpp ../packet10-kopt5/kernels.cpp ../../stage1/packet3-prep/src/sycl_device.cpp)
endif()
get_filename_component(ROOT ${P}/../../../.. ABSOLUTE)
set(UNITS)
foreach(s IN LISTS SOURCES)
 get_filename_component(abs ${R}/${s} ABSOLUTE)
 if(NOT HOST_ONLY)
  file(RELATIVE_PATH rel ${ROOT} ${abs})
  set(abs ${P}/build/checked/${rel})
 endif()
 list(APPEND UNITS ${abs})
endforeach()
function(config t)
 target_include_directories(${t} PRIVATE ${P} ${R} ${O} ${K}/kernels ${K}/loader)
 target_compile_options(${t} PRIVATE -O2 ${PRECISE})
 target_link_options(${t} PRIVATE ${PRECISE})
 target_link_libraries(${t} PRIVATE OpenSSL::Crypto)
 if(HOST_ONLY)
  target_compile_definitions(${t} PRIVATE OWN_RT_HOST_ONLY)
  target_compile_options(${t} PRIVATE -fopenmp)
  target_link_options(${t} PRIVATE -fopenmp)
 else()
  target_compile_options(${t} PRIVATE -fsycl)
  target_link_options(${t} PRIVATE -fsycl)
  target_link_libraries(${t} PRIVATE ze_loader)
 endif()
 if(SANITIZE)
  target_compile_options(${t} PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(${t} PRIVATE -fsanitize=address,undefined)
 endif()
 if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  set_property(TARGET ${t} PROPERTY BUILD_RPATH "/opt/intel/oneapi/compiler/2026.0/lib")
 endif()
endfunction()
add_library(runtime OBJECT ${UNITS})
config(runtime)
add_library(bench_entry OBJECT bench.cpp)
config(bench_entry)
target_compile_definitions(bench_entry PRIVATE BENCH_ENTRY=packet34_entry)
file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/entry.cpp "int packet34_entry(int,char**);int main(int argc,char**argv){return packet34_entry(argc,argv); }\n")
add_executable(replay-bench ${CMAKE_CURRENT_BINARY_DIR}/entry.cpp $<TARGET_OBJECTS:bench_entry> $<TARGET_OBJECTS:runtime>)
config(replay-bench)
if(NOT HOST_ONLY)
 target_compile_definitions(bench_entry PRIVATE P29_MAIN="${P}/build/checked/experiments/own-xpu-runtime/stage2/packet29-merged/main.cpp")
 add_executable(jit-smoke ${R}/jit_smoke.cpp $<TARGET_OBJECTS:bench_entry> $<TARGET_OBJECTS:runtime>)
 config(jit-smoke)
endif()
