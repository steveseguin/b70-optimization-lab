cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet15 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(OWNER ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src)
add_executable(host-test runner.cpp)
target_compile_definitions(host-test PRIVATE HOST_ONLY)
target_compile_options(host-test PRIVATE -O2 -fno-fast-math -ffp-contract=off)
option(NATIVE "Compile only; never discover or run device at build time" OFF)
if(NATIVE)
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Native compilation requires icpx")
 endif()
 add_executable(device-test runner.cpp native.cpp ${OWNER}/owner.cpp ${OWNER}/sycl_device.cpp ../packet4/owner_retire.cpp)
 target_include_directories(device-test PRIVATE ${OWNER})
 target_compile_options(device-test PRIVATE -O2 -fsycl -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
 target_link_options(device-test PRIVATE -fsycl)
 target_link_libraries(device-test PRIVATE ze_loader)
endif()
add_library(math-probe SHARED math_probe.cpp)
target_compile_options(math-probe PRIVATE -O2 -fno-fast-math -ffp-contract=off)
