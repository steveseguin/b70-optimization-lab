cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet52 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "CPU-only preparation" ON)
include(${P52_BASELINE_MANIFEST})
find_package(OpenSSL REQUIRED)
add_executable(freeorder-bench ${P52_SOURCE_MIRROR}/bench.cpp ${P52_BASELINE_OBJECTS})
target_include_directories(freeorder-bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../packet45-merged3 ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/kernels ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/loader)
target_compile_options(freeorder-bench PRIVATE -O2 -fno-fast-math -ffp-contract=off -Werror=return-type)
target_link_options(freeorder-bench PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(freeorder-bench PRIVATE OpenSSL::Crypto)
if(OWN_RT_HOST_ONLY)
 target_compile_definitions(freeorder-bench PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(freeorder-bench PRIVATE -fopenmp)
 target_link_options(freeorder-bench PRIVATE -fopenmp)
else()
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Device build requires IntelLLVM")
 endif()
 target_compile_options(freeorder-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_options(freeorder-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_libraries(freeorder-bench PRIVATE ze_loader)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
 get_filename_component(COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
 set_property(TARGET freeorder-bench PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
endif()
