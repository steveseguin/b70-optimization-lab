get_filename_component(REPO_ROOT ${P}/../../../.. ABSOLUTE)
function(source_path input output)
 get_filename_component(abs ${input} ABSOLUTE BASE_DIR ${P})
 if(OWN_RT_HOST_ONLY)
  set(${output} ${abs} PARENT_SCOPE)
 else()
  file(RELATIVE_PATH rel ${REPO_ROOT} ${abs})
  set(${output} ${P}/build/checked/${rel} PARENT_SCOPE)
 endif()
endfunction()
if(OWN_RT_HOST_ONLY)
 execute_process(COMMAND ${CPU_PYTHON} -B ${P}/validation_mirror.py ${CMAKE_CURRENT_BINARY_DIR}/validation RESULT_VARIABLE VALIDATION)

  message(FATAL_ERROR "Device validation mirror failed")
 endif()
endif()
set(PROGRAM_SOURCES)
foreach(src IN LISTS COMMON)

 list(APPEND PROGRAM_SOURCES ${target_src})
endforeach()
foreach(pair "main|packet26_model_entry" "nongemv_fixture|packet26_fixture_entry" "k2_fixture|packet26_k2_entry")
 string(REPLACE "|" ";" fields ${pair})
 list(GET fields 0 stem)
 list(GET fields 1 entry)
 source_path(${stem}.cpp target_src)
 set(wrapper ${CMAKE_CURRENT_BINARY_DIR}/entry-${stem}.cpp)
 file(WRITE ${wrapper} "#define main ${entry}\n#include \"${target_src}\"\n#undef main\n")
 list(APPEND PROGRAM_SOURCES ${wrapper})
endforeach()
add_library(programs OBJECT ${PROGRAM_SOURCES})
function(configure_program target)
 target_include_directories(${target} PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/validation ${P} ${OWNER} ${P2}/kernels ${P2}/loader)
 target_compile_options(${target} PRIVATE -O2 -Werror=return-type ${PRECISE})
 target_link_options(${target} PRIVATE ${PRECISE})
 target_link_libraries(${target} PRIVATE OpenSSL::Crypto)
 if(OWN_RT_HOST_ONLY)
  target_compile_definitions(${target} PRIVATE OWN_RT_HOST_ONLY)
  target_compile_options(${target} PRIVATE -O3 -march=native -fopenmp)
  target_link_options(${target} PRIVATE -fopenmp)
  if(OWN_RT_SANITIZE)
   target_compile_options(${target} PRIVATE -O1 -g -fsanitize=address,undefined -fno-omit-frame-pointer)
   target_link_options(${target} PRIVATE -fsanitize=address,undefined)
  endif()
 else()
  target_compile_options(${target} PRIVATE -fsycl)
  target_link_options(${target} PRIVATE -fsycl)
  target_link_libraries(${target} PRIVATE ze_loader)
 endif()
 if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  get_filename_component(COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
  set_property(TARGET ${target} PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
 endif()
endfunction()
