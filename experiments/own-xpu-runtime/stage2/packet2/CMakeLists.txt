cmake_minimum_required(VERSION 3.25)
set(CMAKE_CXX_COMPILER "/opt/intel/oneapi/compiler/2026.0/bin/icpx" CACHE FILEPATH "Pinned compiler")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(iq3_packet2 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "2026.0.0")
  message(FATAL_ERROR "Requires pinned icpx 2026.0.0")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(IQ3_DEVICE "Compile/link device test; NEVER registered in CTest" OFF)
set(OWNER ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src)
add_compile_options(-Wall -Wextra -Werror -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
include_directories(loader kernels ${OWNER})
add_library(owner STATIC ${OWNER}/owner.cpp)
add_library(gguf STATIC loader/gguf.cpp)
target_link_libraries(gguf PUBLIC owner)
add_executable(host-tests tests/host.cpp)
target_link_libraries(host-tests PRIVATE gguf)
if(IQ3_DEVICE)
  add_library(kernels STATIC kernels/device.cpp ${OWNER}/sycl_device.cpp)
  target_compile_options(kernels PRIVATE -fsycl)
  add_executable(device-test tests/device.cpp)
  target_compile_options(device-test PRIVATE -fsycl)
  target_link_options(device-test PRIVATE -fsycl)
  target_link_libraries(device-test PRIVATE kernels gguf ze_loader)
endif()
