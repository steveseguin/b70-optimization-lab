cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet53 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "CPU-only preparation" ON)
include(${P53_BASELINE_MANIFEST})
find_package(OpenSSL REQUIRED)
add_executable(expert-order-bench ${P53_SOURCE_MIRROR}/bench.cpp ${P53_BASELINE_OBJECTS})
target_include_directories(expert-order-bench PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../packet45-merged3 ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/kernels ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/loader)
target_compile_options(expert-order-bench PRIVATE -O2 -fno-fast-math -ffp-contract=off -Werror=return-type)
target_link_options(expert-order-bench PRIVATE -fno-fast-math -ffp-contract=off)
target_link_libraries(expert-order-bench PRIVATE OpenSSL::Crypto)
if(OWN_RT_HOST_ONLY)
 target_compile_definitions(expert-order-bench PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(expert-order-bench PRIVATE -fopenmp)
 target_link_options(expert-order-bench PRIVATE -fopenmp)
else()
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Device build requires IntelLLVM")
 endif()
 target_compile_options(expert-order-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_options(expert-order-bench PRIVATE -fsycl -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_link_libraries(expert-order-bench PRIVATE ze_loader)
endif()
if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
 get_filename_component(COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
 set_property(TARGET expert-order-bench PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
endif()

if(OWN_RT_HOST_ONLY)
 add_executable(cpu-identity-test ${P53_SOURCE_MIRROR}/cpu_identity_test.cpp)
 target_include_directories(cpu-identity-test PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/../packet45-merged3 ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/kernels ${CMAKE_CURRENT_SOURCE_DIR}/../packet2/loader)
 target_compile_definitions(cpu-identity-test PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(cpu-identity-test PRIVATE -O2 -fno-fast-math -ffp-contract=off -fopenmp -Werror=return-type)
 target_link_options(cpu-identity-test PRIVATE -fno-fast-math -ffp-contract=off -fopenmp)
 target_link_libraries(cpu-identity-test PRIVATE OpenSSL::Crypto)
 if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  set_property(TARGET cpu-identity-test PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
 endif()
endif()
