cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(own_rt_iq3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "CPU mock only, without accelerator libraries" ON)
set(OWNER ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src)
set(P2 ${CMAKE_CURRENT_SOURCE_DIR}/../packet2)
find_package(OpenSSL REQUIRED)
add_executable(own-rt-iq3 main.cpp kernels.cpp owner_retire.cpp ${P2}/loader/gguf.cpp ${OWNER}/owner.cpp)
target_include_directories(own-rt-iq3 PRIVATE ${OWNER} ${P2}/kernels ${P2}/loader)
target_compile_options(own-rt-iq3 PRIVATE -O2 -Wall -Wextra -fno-fast-math -ffp-contract=off)
target_link_libraries(own-rt-iq3 PRIVATE OpenSSL::Crypto)
if(OWN_RT_HOST_ONLY)
 target_compile_definitions(own-rt-iq3 PRIVATE OWN_RT_HOST_ONLY)
else()
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Use the installed icpx for the compile-only SYCL target")
 endif()
 target_sources(own-rt-iq3 PRIVATE ${P2}/kernels/device.cpp ${OWNER}/sycl_device.cpp)
 target_compile_options(own-rt-iq3 PRIVATE -fsycl -fdenormal-fp-math=ieee)
 target_link_options(own-rt-iq3 PRIVATE -fsycl)
 target_link_libraries(own-rt-iq3 PRIVATE ze_loader)
endif()
# Deliberately no native CTest or configure-time accelerator discovery.
if(OWN_RT_HOST_ONLY)
 add_executable(owner-retirement-tests owner_test.cpp owner_retire.cpp ${OWNER}/owner.cpp)
 target_include_directories(owner-retirement-tests PRIVATE ${OWNER})
endif()
