cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(packet20fusion LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(PRECISION -fno-fast-math -ffp-contract=off)
if(CMAKE_CXX_COMPILER_ID MATCHES "IntelLLVM|Clang")
 list(APPEND PRECISION -fdenormal-fp-math=ieee)
endif()
add_executable(fusion-plan plan.cpp)
add_executable(fusion-host gate.cpp)
target_compile_options(fusion-host PRIVATE -O2 -mfma ${PRECISION})
target_link_options(fusion-host PRIVATE ${PRECISION})
if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
 add_executable(fusion-device device.cpp)
 set(DEVICE_PRECISION ${PRECISION} -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_compile_options(fusion-device PRIVATE -O2 -fsycl ${DEVICE_PRECISION})
 target_link_options(fusion-device PRIVATE -fsycl ${DEVICE_PRECISION})
endif()
