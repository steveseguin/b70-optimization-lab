cmake_minimum_required(VERSION 3.25)
set(CMAKE_CXX_COMPILER "/opt/intel/oneapi/compiler/2026.0/bin/icpx" CACHE FILEPATH "Pinned compiler")
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(iq3_packet6 LANGUAGES CXX)
if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM" OR NOT CMAKE_CXX_COMPILER_VERSION VERSION_EQUAL "2026.0.0")
  message(FATAL_ERROR "Requires installed icpx 2026.0.0")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BASE ${CMAKE_CURRENT_SOURCE_DIR}/../packet2)
set(OWNER ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src)
add_compile_options(-Wall -Wextra -Werror -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee)
include_directories(${CMAKE_CURRENT_SOURCE_DIR} ${BASE}/kernels ${BASE}/tests ${OWNER})
add_library(owner STATIC ${OWNER}/owner.cpp)
add_library(kopt STATIC kernels.cpp ${BASE}/kernels/device.cpp ${OWNER}/sycl_device.cpp)
target_compile_options(kopt PRIVATE -fsycl)
foreach(target device-test device-timing)
  if(target STREQUAL "device-test")
    set(source tests/device.cpp)
  else()
    set(source tests/timing.cpp)
  endif()
  add_executable(${target} ${source})
  target_compile_options(${target} PRIVATE -fsycl)
  target_link_options(${target} PRIVATE -fsycl)
  target_link_libraries(${target} PRIVATE kopt owner ze_loader)
endforeach()
add_executable(host-tests tests/host.cpp)
