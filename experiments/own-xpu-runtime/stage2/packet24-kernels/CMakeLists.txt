cmake_minimum_required(VERSION 3.25)
set(CMAKE_TRY_COMPILE_TARGET_TYPE STATIC_LIBRARY)
project(own_rt_iq3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(OWN_RT_HOST_ONLY "CPU mock only, without accelerator libraries" ON)
set(OWNER ${CMAKE_CURRENT_SOURCE_DIR}/../../stage1/packet3-prep/src)
set(P2 ${CMAKE_CURRENT_SOURCE_DIR}/../packet2)
find_package(OpenSSL REQUIRED)
find_program(CPU_PYTHON NAMES python3 REQUIRED)
execute_process(COMMAND ${CPU_PYTHON} -B ${CMAKE_CURRENT_SOURCE_DIR}/math_gate.py RESULT_VARIABLE MATH_GATE OUTPUT_QUIET)
if(NOT MATH_GATE EQUAL 0)
 message(FATAL_ERROR "Device platform-math grep gate failed")
endif()
add_custom_target(math-gate COMMAND ${CPU_PYTHON} -B ${CMAKE_CURRENT_SOURCE_DIR}/math_gate.py
 COMMAND ${CPU_PYTHON} -B ${CMAKE_CURRENT_SOURCE_DIR}/queue_gate.py)

add_executable(own-rt-iq3 main.cpp gemv_batch.cpp performance.cpp adapter.cpp runtime_kernels.cpp owner_retire.cpp ${P2}/loader/gguf.cpp owner.cpp)
add_dependencies(own-rt-iq3 math-gate)
target_include_directories(own-rt-iq3 PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER} ${P2}/kernels ${P2}/loader)
target_compile_options(own-rt-iq3 PRIVATE -O2 -Wall -Wextra -fno-fast-math -ffp-contract=off)
target_link_libraries(own-rt-iq3 PRIVATE OpenSSL::Crypto)
if(OWN_RT_HOST_ONLY)
 target_compile_definitions(own-rt-iq3 PRIVATE OWN_RT_HOST_ONLY)
 target_compile_options(own-rt-iq3 PRIVATE -O3 -march=native -fopenmp)
 target_link_options(own-rt-iq3 PRIVATE -fopenmp -fno-fast-math -ffp-contract=off)
else()
 if(NOT CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
  message(FATAL_ERROR "Use the installed icpx for the compile-only SYCL target")
 endif()
 target_sources(own-rt-iq3 PRIVATE ${P2}/kernels/device.cpp ${OWNER}/sycl_device.cpp ../packet6-kopt/kernels.cpp ../packet7-kopt2/kernels.cpp ../packet8-kopt3/kernels.cpp ../packet9-kopt4/kernels.cpp ../packet10-kopt5/kernels.cpp native.cpp)
 set(PRECISE -fno-fast-math -ffp-contract=off -fdenormal-fp-math=ieee -foffload-fp32-prec-div -foffload-fp32-prec-sqrt)
 target_compile_options(own-rt-iq3 PRIVATE -fsycl ${PRECISE})
 target_link_options(own-rt-iq3 PRIVATE -fsycl ${PRECISE})
 target_link_libraries(own-rt-iq3 PRIVATE ze_loader)
 # Loader-only preflight must work without sourcing a shell-wide oneAPI setup.
 get_filename_component(COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
 set_property(TARGET own-rt-iq3 PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
endif()
# Deliberately no native CTest or configure-time accelerator discovery.
if(OWN_RT_HOST_ONLY)
 add_executable(authority-tests authority_test.cpp)
 target_compile_options(authority-tests PRIVATE -O2 -fno-fast-math -ffp-contract=off)
 add_executable(owner-retirement-tests ../packet4/owner_test.cpp owner_retire.cpp owner.cpp)
 target_include_directories(owner-retirement-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER})
endif()

if(OWN_RT_HOST_ONLY)
 add_executable(chain-tests chain_test.cpp owner_retire.cpp owner.cpp)
 target_include_directories(chain-tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER})
endif()

if(OWN_RT_HOST_ONLY)
 add_library(math_bridge SHARED math_bridge.cpp)

if(OWN_RT_HOST_ONLY)
 add_executable(queue-tests queue_test.cpp owner_retire.cpp owner.cpp)

# Fixture programs are only executed from CPU builds during preparation.
add_executable(fused-fixtures gemv_fixture.cpp gemv_batch.cpp performance.cpp owner.cpp owner_retire.cpp)
add_executable(nongemv-fixtures nongemv_fixture.cpp route_fixture.cpp kernels_fixture.cpp adapter_fixture.cpp adapter.cpp runtime_kernels.cpp owner.cpp owner_retire.cpp)
foreach(fixture fused-fixtures nongemv-fixtures)
 target_include_directories(${fixture} PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER} ${P2}/kernels ${P2}/loader)
 target_compile_options(${fixture} PRIVATE -O3 -mfma -fno-fast-math -ffp-contract=off)
 target_link_libraries(${fixture} PRIVATE OpenSSL::Crypto)
 if(OWN_RT_HOST_ONLY)
  target_compile_definitions(${fixture} PRIVATE OWN_RT_HOST_ONLY)
  target_compile_options(${fixture} PRIVATE -fopenmp)
  target_link_options(${fixture} PRIVATE -fopenmp -fno-fast-math -ffp-contract=off)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "IntelLLVM")
   get_filename_component(HOST_COMPILER_BIN ${CMAKE_CXX_COMPILER} DIRECTORY)
   set_property(TARGET ${fixture} PROPERTY BUILD_RPATH "${HOST_COMPILER_BIN}/../lib")
  endif()
 else()
  target_sources(${fixture} PRIVATE ${OWNER}/sycl_device.cpp)
  target_compile_options(${fixture} PRIVATE -fsycl ${PRECISE})
  target_link_options(${fixture} PRIVATE -fsycl ${PRECISE})
  target_link_libraries(${fixture} PRIVATE ze_loader)
  set_property(TARGET ${fixture} PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
 endif()
endforeach()
if(NOT OWN_RT_HOST_ONLY)
 target_sources(nongemv-fixtures PRIVATE native.cpp)

  set_property(TARGET k2-fixtures PROPERTY ${prop} "${value}")

if(NOT OWN_RT_HOST_ONLY)
 execute_process(COMMAND ${CPU_PYTHON} -B ${CMAKE_CURRENT_SOURCE_DIR}/kernel_gate.py RESULT_VARIABLE CAPTURE_GATE)
 if(NOT CAPTURE_GATE EQUAL 0)
  message(FATAL_ERROR "Kernel wrapper gate failed")
 endif()
 get_filename_component(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../../.. ABSOLUTE)
 foreach(t own-rt-iq3 fused-fixtures nongemv-fixtures k2-fixtures)
  get_target_property(srcs ${t} SOURCES)
  set(checked_srcs)
  foreach(src IN LISTS srcs)
   get_filename_component(abs ${src} ABSOLUTE BASE_DIR ${CMAKE_CURRENT_SOURCE_DIR})
   file(RELATIVE_PATH rel ${REPO_ROOT} ${abs})
   list(APPEND checked_srcs ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/${rel})
  endforeach()
  set_property(TARGET ${t} PROPERTY SOURCES ${checked_srcs})
 endforeach()
endif()

if(NOT OWN_RT_HOST_ONLY)
 get_target_property(main_sources own-rt-iq3 SOURCES)
 set(jit_sources)
 foreach(src IN LISTS main_sources)
  if(src MATCHES "/main.cpp$")
   file(READ ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/k2.cpp content)
   string(REPLACE "int main(int argc,char** argv)" "int packet24_k2_entry(int argc,char** argv)" content "${content}")
   set(dst ${CMAKE_CURRENT_SOURCE_DIR}/build/jit-main.cpp)
   file(WRITE ${dst} "${content}")
   list(APPEND jit_sources ${dst})
  else()
   list(APPEND jit_sources ${src})
  endif()
 endforeach()
 foreach(pair gemv_fixture nongemv_fixture)
  set(src ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/${pair}.cpp)
  file(READ ${src} content)
  string(REPLACE "int main(int argc,char** argv)" "int packet23_${pair}_entry(int argc,char** argv)" content "${content}")
  file(WRITE ${CMAKE_CURRENT_SOURCE_DIR}/build/jit-${pair}.cpp "${content}")
  list(APPEND jit_sources ${CMAKE_CURRENT_SOURCE_DIR}/build/jit-${pair}.cpp)
 endforeach()
 list(APPEND jit_sources ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/adapter_fixture.cpp ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/kernels_fixture.cpp ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/route_fixture.cpp)
 add_custom_target(kernel-gate COMMAND ${CPU_PYTHON} -B ${CMAKE_CURRENT_SOURCE_DIR}/kernel_gate.py)
 add_library(packet23-programs OBJECT ${jit_sources})
 add_dependencies(packet23-programs kernel-gate)
 target_include_directories(packet23-programs PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER} ${P2}/kernels ${P2}/loader)
 target_compile_options(packet23-programs PRIVATE -O2 -fsycl ${PRECISE})
 add_executable(jit-smoke ${CMAKE_CURRENT_SOURCE_DIR}/build/checked/experiments/own-xpu-runtime/stage2/packet24-kernels/jit_smoke.cpp $<TARGET_OBJECTS:packet23-programs>)
 # Every native executable links exactly the same object/image set. K0 builds
 # all of these kernels without calling any program entrypoint.
 foreach(pair "own-rt-iq3|packet23_model_entry" "fused-fixtures|packet23_gemv_fixture_entry" "nongemv-fixtures|packet23_nongemv_fixture_entry" "k2-fixtures|packet24_k2_entry")
  string(REPLACE "|" ";" values ${pair})
  list(GET values 0 target)
  list(GET values 1 entry)
  set(shim ${CMAKE_CURRENT_SOURCE_DIR}/build/${target}-entry.cpp)
  file(WRITE ${shim} "int ${entry}(int,char**); int main(int argc,char**argv){return ${entry}(argc,argv);}\n")
  set_property(TARGET ${target} PROPERTY SOURCES ${shim} $<TARGET_OBJECTS:packet23-programs>)
 endforeach()
 target_include_directories(jit-smoke PRIVATE ${CMAKE_CURRENT_SOURCE_DIR} ${OWNER} ${P2}/kernels ${P2}/loader)
 target_compile_options(jit-smoke PRIVATE -O2 -fsycl ${PRECISE})
 target_link_options(jit-smoke PRIVATE -fsycl ${PRECISE})
 target_link_libraries(jit-smoke PRIVATE OpenSSL::Crypto ze_loader)
 set_property(TARGET jit-smoke PROPERTY BUILD_RPATH "${COMPILER_BIN}/../lib")
endif()

# CPU executables call explicitly renamed entry functions, exactly like native
# shims. Every regular CPU gate therefore checks the formerly missing return.
if(OWN_RT_HOST_ONLY)
 foreach(pair "own-rt-iq3|main|packet23_model_entry" "fused-fixtures|gemv_fixture|packet23_gemv_fixture_entry" "nongemv-fixtures|nongemv_fixture|packet23_nongemv_fixture_entry" "k2-fixtures|k2|packet24_k2_entry")
  string(REPLACE "|" ";" values ${pair})
  list(GET values 0 target)
  list(GET values 1 stem)
  list(GET values 2 entry)
  set(src ${CMAKE_CURRENT_SOURCE_DIR}/${stem}.cpp)
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${src})
  file(READ ${src} text)
  string(REPLACE "int main(int argc, char **argv)" "int ${entry}(int argc, char **argv)" text "${text}")
  string(REPLACE "int main(int argc,char** argv)" "int ${entry}(int argc,char** argv)" text "${text}")
  set(dst ${CMAKE_CURRENT_SOURCE_DIR}/build/host-entry-${stem}.cpp)
  file(WRITE ${dst} "#line 1 \"${src}\"\n${text}")
  set(shim ${CMAKE_CURRENT_SOURCE_DIR}/build/host-${target}-entry.cpp)
  file(WRITE ${shim} "int ${entry}(int,char**); int main(int argc,char**argv){return ${entry}(argc,argv); }\n")
  get_target_property(sources ${target} SOURCES)
  list(REMOVE_ITEM sources ${stem}.cpp)
  set_property(TARGET ${target} PROPERTY SOURCES ${dst} ${shim} ${sources})
 endforeach()
endif()
