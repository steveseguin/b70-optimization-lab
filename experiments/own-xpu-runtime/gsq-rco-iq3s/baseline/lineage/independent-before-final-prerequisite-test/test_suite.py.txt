class SuiteTests(unittest.TestCase):
 def fixture(self,root,token_bad=False,state_bad=False,throw_at=None):
  t=types.SimpleNamespace(out=Path(root),children=[types.SimpleNamespace(pid=11),types.SimpleNamespace(pid=12)],calls=[],resets=[],checks=0)
  t.check_alive=lambda:setattr(t,'checks',t.checks+1)
  def reset(t,d):d.mkdir();t.request_output=d;t.resets.append(str(d));t.fresh=True
  class Controller:
   def __init__(self,t,draft,*args):self.t=t;self.mode=draft.mode
   def run(self,prompt,cap,eos):
    if not self.t.fresh:raise AssertionError('request without full reset')
    self.t.fresh=False;self.t.calls.append((self.mode,list(prompt),cap,tuple(eos)));self.t.mode=self.mode
    if len(self.t.calls)==throw_at:raise RuntimeError('original provider error')
    tokens=[3 if token_bad and self.mode=='mtp' and cap==128 else 2]*cap
    return dict(tokens=tokens,timing_rows=[dict(generated_index=i+1,token_id=v,completed_monotonic=100+i*.1) for i,v in enumerate(tokens)],decode_wall_ms=(cap-1)*100,decode_intervals=cap-1,prompt_wall_ms=5,stats=dict(guesses=cap-1 if self.mode=='mtp' else 0,hits=cap-1 if self.mode=='mtp' else 0,misses=0))
  state=lambda t:{'0':{'state':'wrong' if state_bad and t.mode=='mtp' else 'exact'},'1':{'state':'exact'}}
  return t,reset,Controller,state
 def run_fake(self,root,**kw):

   return t,S.run_attached(t,plan)
