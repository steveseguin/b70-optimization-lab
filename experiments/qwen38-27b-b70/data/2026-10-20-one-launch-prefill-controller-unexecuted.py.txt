#!/usr/bin/env python3
"""One FP8 overlay-0.2.0 server, one synthetic prefill screen, no retry.

Default/--plan is CPU-only and writes nothing. --run requires a systemd user
unit. --serve-adapter is an internal child: unchanged package start(), with
only stop_owned replaced to prohibit Docker's timed SIGKILL escalation.
"""
import argparse
from contextlib import ExitStack
import datetime as dt
import fcntl
import hashlib
import importlib.util
import json
import os
from pathlib import Path
import signal
import shutil
import socket
import subprocess
import sys
import time

ROOT = Path(__file__).resolve().parents[3]
SELF = Path(__file__).resolve()
PREREG = ROOT / 'experiments/qwen38-27b-b70/data/2026-10-20-one-launch-prefill-prereg.json'
SERVE = ROOT / 'packages/qwen38-27b-fp8-tp2-b70/scripts/serve.py'
BENCH = SELF.with_name('bench-short-prefill.py')
MODEL = Path('/mnt/fast-ai/llm-models/qwen3.8-27b-fp8')
OUT = Path('/mnt/usb-models/bench-results/fp8-one-launch-prefill-20261020')
HEALTH = ROOT / 'scripts/check-qwen36-xpu-xccl-health.sh'
PYTHON_XPU = Path('/home/steve/.venvs/vllm-xpu/bin/python')
LOCKS = ('/tmp/b70-benchmark.lock', '/tmp/b70-gpu0.lock', '/tmp/b70-gpu1.lock')
DEVICES = {'renderD130': '0000:23:00.0', 'renderD131': '0000:27:00.0'}
PORT = 18124


def now():
    return dt.datetime.now(dt.timezone.utc).isoformat()


def dump(path, data):
    path.write_text(json.dumps(data, indent=2, ensure_ascii=False) + '\n')


def load_serve():
    spec = importlib.util.spec_from_file_location('fp8_package_serve', SERVE)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def verify_pins():
    prereg = json.loads(PREREG.read_text())
    for item in prereg['source_files']:
        actual = hashlib.sha256((ROOT / item['path']).read_bytes()).hexdigest()
        if actual != item['sha256']:
            raise RuntimeError('Preregistered source changed: ' + item['path'])
    overlays = sorted(str(p.relative_to(ROOT)) for p in (SERVE.parent.parent / 'overlays').rglob('*')
                      if p.is_file() and '__pycache__' not in p.parts)
    if overlays != prereg['overlay_files']:
        raise RuntimeError('Overlay inventory changed; refusing unpinned runtime input')
    return prereg


def client_argv(out):
    return [sys.executable, str(BENCH), '--base-url', f'http://127.0.0.1:{PORT}',
            '--model', 'qwen38-27b-fp8', '--out', str(out / 'screen'),
            '--lengths', '512', '--repeats', '2', '--max-tokens', '128',
            '--max-model-len', '33024', '--timeout', '120']


def guarded_client(out):
    """Original client and requests; a passive fault check before every fetch."""
    verify_pins()
    spec = importlib.util.spec_from_file_location('prefill_client', BENCH)
    client = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(client)
    module = load_serve()
    original_fetch = client.fetch
    since = os.environ['PREFILL_MONITOR_SINCE']

    def fetch(*args, **kwargs):
        disk_floors(out)
        kernel = module.journal(since)
        (out / 'client-guard-kernel.log').write_text(kernel)
        bad = [line for line in kernel.splitlines() if module.FAULT.search(line)]
        if bad:
            fault = out / 'FAULT.json'
            if not fault.exists():
                dump(fault, {'at': now(), 'lines': bad, 'new_requests_halted': True,
                             'detector': 'per-request-client-guard'})
            raise RuntimeError('Kernel fault; client admission halted')
        return original_fetch(*args, **kwargs)

    client.fetch = fetch
    client.run(argparse.Namespace(base_url=f'http://127.0.0.1:{PORT}', model='qwen38-27b-fp8',
        out=out / 'screen', baseline=None, lengths='512', repeats=2, max_tokens=128,
        max_model_len=33024, timeout=120))


def plan(out):
    module = load_serve()
    prereg = verify_pins()
    state = out / 'server'
    return {'scope': prereg['scope'], 'preregistration': str(PREREG),
            'preregistration_sha256': hashlib.sha256(PREREG.read_bytes()).hexdigest(),
            'source_files': prereg['source_files'], 'host': 'steve-b70s',
            'device_pci_mapping': DEVICES, 'locks': list(LOCKS), 'out': str(out),
            'launch_argv_template': module.docker_argv('recommended', MODEL, state, PORT, 'OWNED-UUID'),
            'client_argv': client_argv(out),
            'guarded_client_argv': [sys.executable, str(SELF), '--client-adapter', '--out', str(out)],
            'client_adapter': 'Passive journal check before every HTTP admission; adds overhead to HTTP proxy only',
            'lifecycle_adapter': 'TERM only; owned container; no timed KILL',
            'bounds_seconds': {'model_verification': 1200, 'health': 180, 'startup': 1800,
                               'screen': 1200, 'graceful_container_wait': 180, 'adapter_wait': 240},
            'disk_free_floors_gib': {'root': 5, 'output': 30},
            'image_policy': 'verify cached pinned image before active probes; no pull/build',
            'active_health': {'script': str(HEALTH), 'physical_devices': '0,1',
                              'xccl_devices': '0,1', 'timeout_s': 90,
                              'policy': 'each device once, then XCCL once; abort before next on failure/fault; same after success only'},
            'no_quality_loss_gate': False, 'strict_headline_eligible': False}


def disk_floors(out):
    # Run-only floors: cached image already exists; no build or pull. New
    # vLLM cache and all output use the separately mounted output filesystem.
    if shutil.disk_usage(ROOT).free < 5 * 1024**3:
        raise RuntimeError('Root filesystem below run-only 5 GiB floor; halt without deleting artifacts')
    if shutil.disk_usage(out if out.exists() else out.parent).free < 30 * 1024**3:
        raise RuntimeError('Output filesystem below 30 GiB floor; halt without deleting artifacts')


def term_only_stop(module, state, record, *, clock=time.monotonic, sleep=time.sleep):
    """Same ownership receipt protocol, but TERM plus passive inspection only."""
    with (state / 'stop.lock').open('a') as lock:
        fcntl.flock(lock, fcntl.LOCK_EX)
        receipt = state / 'stop-request.json'
        if receipt.exists():
            return
        identity = record.get('container_id')
        if not identity:
            raise RuntimeError('No captured container identity; refusing any container signal')
        container = module.inspect_container(identity)
        if container is None:
            dump(receipt, {'at': now(), 'container_id': identity, 'already_absent': True})
            return
        module.owned(record, container)
        dump(receipt, {'at': now(), 'container_id': identity, 'requested': True,
                       'signal': 'SIGTERM', 'escalation': 'none', 'wait_seconds': 180})
        if container['State']['Running']:
            result = module.run(['docker', 'kill', '--signal=TERM', identity], check=False, timeout=30)
            (state / 'stop.log').write_text(result.stdout + result.stderr)
            if result.returncode:
                raise RuntimeError('Owned TERM request failed; no signal retry')
        deadline = clock() + 180
        while clock() < deadline:
            container = module.inspect_container(identity)
            if container is None or not container['State']['Running']:
                dump(state / 'graceful-stop.json', {'at': now(), 'stopped': True, 'escalation': False})
                return
            module.owned(record, container)
            sleep(2)
        dump(state / 'graceful-stop.json', {'at': now(), 'stopped': False, 'escalation': False,
                                          'operator_action_required': True})
        raise RuntimeError('Owned container still running after TERM; halted without KILL or restart')


def adapter(out):
    verify_pins()
    module = load_serve()
    module.stop_owned = lambda state, record: term_only_stop(module, state, record)
    original_warmup = module.warm_up
    def guarded_warmup(port):
        disk_floors(out)
        if module.FAULT.search(module.journal(os.environ['PREFILL_MONITOR_SINCE'])):
            raise RuntimeError('Kernel fault before startup warmup; no request admitted')
        return original_warmup(port)
    module.warm_up = guarded_warmup
    args = argparse.Namespace(model_dir=MODEL, state_dir=out / 'server', profile='recommended',
                              port=PORT, startup_timeout=1800)
    module.start(args)


class Session:
    def __init__(self, out, lock_fds):
        self.out, self.lock_fds = out, lock_fds
        self.module = load_serve()
        self.start = now()
        self.server = self.child = None
        self.interrupted = False
        self.interrupt_count = 0
        self.fault = False
        self.postflight_ok = False
        self.env = self.module.clean_env()
        self.env['B70_FP8_RENDER_NODES'] = ','.join(DEVICES)
        self.env['INVOCATION_ID'] = os.environ.get('INVOCATION_ID', '')
        self.env['PREFILL_MONITOR_SINCE'] = self.start

    def journal(self, label, since=None):
        text = self.module.journal(since or self.start)
        (self.out / f'{label}-kernel.log').write_text(text)
        faults = [line for line in text.splitlines() if self.module.FAULT.search(line)]
        if faults:
            self.fault = True
            dump(self.out / 'FAULT.json', {'at': now(), 'lines': faults, 'new_requests_halted': True})
            raise RuntimeError('Kernel fault: no further requests or active probes')
        disk_floors(self.out)
        if self.interrupted:
            raise RuntimeError('Coordinator interrupted; no further requests')

    def snapshot(self, label):
        dump(self.out / f'{label}-host.json', {'at': now(), 'hostname': socket.gethostname(),
            'uname': list(os.uname()),
            'disk_free_bytes': {'root': shutil.disk_usage(ROOT).free, 'output': shutil.disk_usage(self.out).free},
            'boot_id': Path('/proc/sys/kernel/random/boot_id').read_text().strip(),
            'meminfo': Path('/proc/meminfo').read_text(), 'swaps': Path('/proc/swaps').read_text(),
            'device_mapping': {node: (Path('/sys/class/drm') / node / 'device').resolve().name for node in DEVICES},
            'memory_blocks_53_57': {str(i): (Path(f'/sys/devices/system/memory/memory{i}/state').read_text().strip()
                                          if Path(f'/sys/devices/system/memory/memory{i}/state').exists() else 'absent')
                                    for i in range(53, 58)}})

    def spawn(self, argv, label, extra=None):
        dump(self.out / f'{label}-command.json', {'argv': list(map(str, argv)), 'explicit_env': extra or {}})
        with (self.out / f'{label}.log').open('x') as log:
            return subprocess.Popen(list(map(str, argv)), cwd=ROOT, env={**self.env, **(extra or {})},
                stdout=log, stderr=subprocess.STDOUT, start_new_session=True, pass_fds=self.lock_fds)

    def monitor(self, proc, seconds, label):
        deadline = time.monotonic() + seconds
        while proc.poll() is None:
            self.journal(label)
            if self.server is not None and self.server.poll() is not None:
                raise RuntimeError('Server exited; no client retry')
            if time.monotonic() >= deadline:
                raise RuntimeError(f'{label} exceeded its bound')
            time.sleep(2)
        self.journal(label)
        if proc.returncode:
            raise RuntimeError(f'{label} failed with code {proc.returncode}')

    def phase(self, argv, seconds, label, extra=None):
        self.journal('before-' + label)
        self.child = self.spawn(argv, label, extra)
        try:
            self.monitor(self.child, seconds, label)
        except Exception:
            self.soft_stop(self.child, 20)
            raise
        self.child = None

    @staticmethod
    def soft_stop(proc, seconds):
        if proc is None or proc.poll() is not None:
            return True
        if not getattr(proc, '_soft_stop_sent', False):
            try:
                os.killpg(proc.pid, signal.SIGTERM)
            except ProcessLookupError:
                return True
            proc._soft_stop_sent = True
        try:
            proc.wait(timeout=seconds)
        except subprocess.TimeoutExpired:
            return False
        return True

    def health(self, label):
        self.journal('before-' + label)
        self.module.check_available(PORT, 'prefill-health-ownership-check')
        # The legacy shell continues after a failed probe. Restrict each call
        # to ONE device and gate the next call here; no later work on failure.
        for device in (0, 1):
            self.phase(['bash', HEALTH], 45, f'{label}-device{device}',
                       {'PYTHON': str(PYTHON_XPU), 'PHYSICAL_DEVICES': str(device),
                        'XPU_HEALTH_SKIP_XCCL': '1'})
        self.phase([PYTHON_XPU, '-m', 'torch.distributed.run', '--master-addr=127.0.0.1',
                    '--master-port=29524', '--nproc_per_node=2', ROOT / 'tools/xccl_probe.py', 'allreduce'],
                   90, f'{label}-xccl',
                   {'ONEAPI_DEVICE_SELECTOR': 'level_zero:0,1', 'ZE_AFFINITY_MASK': '0,1',
                    'CCL_ATL_TRANSPORT': 'ofi', 'CCL_TOPO_P2P_ACCESS': '1',
                    'FI_TCP_IFACE': 'lo', 'CCL_KVS_IFACE': 'lo'})

    def owned_container_running(self):
        path = self.out / 'server/state.json'
        if not path.exists():
            return False
        record = json.loads(path.read_text())
        identity = record.get('container_id') or record.get('container_name')
        container = self.module.inspect_container(identity) if identity else None
        if container is None:
            return False
        if record.get('container_id'):
            self.module.owned(record, container)
        return container['State']['Running']

    def execute(self):
        error = None
        client_ok = False
        cleanup = {}
        try:
            self.snapshot('before')
            self.journal('before-recent', since='10 minutes ago')
            self.module.check_available(PORT, 'prefill-preflight-ownership-check')
            dump(self.out / 'local-image.json', {'local_image_id': self.module.local_image_id(),
                                                'registry_image': self.module.IMAGE, 'pull_attempted': False})
            verify_pins()
            self.phase([sys.executable, ROOT / 'repro/qwen38-27b-autoround-int4-b70/scripts/verify-model-direct.py',
                        ROOT / 'repro/qwen38-27b-fp8-vllm-tp2-asrock-b70/model-direct.json', MODEL,
                        '--json', self.out / 'model-verification.json'],
                       1200, 'model-verification')
            self.health('preflight-health')
            self.server = self.spawn([sys.executable, SELF, '--serve-adapter', '--out', self.out], 'launcher')
            deadline = time.monotonic() + 1845
            while True:
                self.journal('startup')
                if self.server.poll() is not None:
                    raise RuntimeError('Server launcher exited before ready; no retry')
                state = self.out / 'server/state.json'
                if state.exists():
                    record = json.loads(state.read_text())
                    if record.get('status') == 'ready':
                        actual = json.loads((self.out / 'server/launch.json').read_text())['argv']
                        expected = self.module.docker_argv('recommended', MODEL, self.out / 'server', PORT,
                                                           record['container_name'])
                        if actual != expected:
                            raise RuntimeError('Runtime launch argv differs from package identity')
                        break
                    if record.get('status') in ('failed', 'stopped'):
                        raise RuntimeError('Server reported failure; no retry')
                if time.monotonic() >= deadline:
                    raise RuntimeError('Server readiness bound exceeded')
                time.sleep(2)
            self.journal('ready')
            if not self.module.healthy(PORT):
                raise RuntimeError('Ready endpoint failed passive health check')
            verify_pins()
            self.phase([sys.executable, SELF, '--client-adapter', '--out', self.out], 1200, 'client')
            client_ok = json.loads((self.out / 'screen/summary.json').read_text()).get('passed') is True
            if not client_ok:
                raise RuntimeError('Client did not pass its diagnostic checks')
        except Exception as exc:
            error = str(exc)
        finally:
            cleanup['client_exited'] = self.soft_stop(self.child, 20)
            cleanup['launcher_exited'] = self.soft_stop(self.server, 240)
            self.server = None
            try:
                self.snapshot('after')
                self.journal('after')
                self.module.check_available(PORT, 'prefill-postflight-ownership-check')
                if not all(cleanup.values()):
                    raise RuntimeError('A child did not drain; no active health probe')
                # Never add GPU work following any failed workload or fault.
                if error is None and not self.fault:
                    self.health('postflight-health')
                    self.postflight_ok = True
            except Exception as exc:
                error = error or str(exc)
            cleanup['last_client_exited'] = self.soft_stop(self.child, 20)
            try:
                cleanup['owned_container_stopped'] = not self.owned_container_running()
            except Exception as exc:
                cleanup['owned_container_stopped'] = False
                error = error or str(exc)
            halted_with_locks = not all(cleanup.values())
            dump(self.out / 'result.json', {'at': now(), 'passed': client_ok and self.postflight_ok and error is None,
                'scope': 'synthetic 512-token prefill screen; ignore_eos; no headline/no-loss gate',
                'error': error, 'fault': self.fault, 'cleanup': cleanup, 'postflight_passed': self.postflight_ok,
                'strict_headline_eligible': False, 'no_quality_loss_gate': False,
                'server_attempts': 1 if (self.out / 'launcher-command.json').exists() else 0,
                'no_retry': True, 'no_sigkill': True, 'holding_locks_for_operator': halted_with_locks})
            if halted_with_locks:
                count = self.interrupt_count
                dump(self.out / 'HALTED.json', {'at': now(), 'reason': 'owned work did not drain',
                    'locks_held': True, 'new_work_admission_stopped': True,
                    'release': 'operator SIGTERM/SIGINT to this controller; does not kill the remaining container'})
                print('HALTED: no more active work; holding locks until operator signals this controller', flush=True)
                while self.interrupt_count == count:
                    signal.pause()
        if error:
            raise RuntimeError(error)


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    mode = parser.add_mutually_exclusive_group()
    mode.add_argument('--plan', action='store_true')
    mode.add_argument('--run', action='store_true')
    mode.add_argument('--serve-adapter', action='store_true', help=argparse.SUPPRESS)
    mode.add_argument('--client-adapter', action='store_true', help=argparse.SUPPRESS)
    parser.add_argument('--out', type=Path, default=OUT)
    args = parser.parse_args()
    out = args.out.resolve()
    if args.serve_adapter or args.client_adapter:
        if socket.gethostname() != 'steve-b70s' or not os.environ.get('INVOCATION_ID'):
            raise RuntimeError('Internal adapter requires the reviewed systemd session on steve-b70s')
        if args.serve_adapter:
            adapter(out)
        else:
            guarded_client(out)
        return
    identity = plan(out)
    if not args.run:
        print(json.dumps(identity, indent=2))
        return
    if socket.gethostname() != 'steve-b70s' or not os.environ.get('INVOCATION_ID'):
        raise RuntimeError('Run only in a systemd user unit on steve-b70s')
    mem = dict(line.split(':', 1) for line in Path('/proc/meminfo').read_text().splitlines())
    if int(mem['MemAvailable'].split()[0]) < 16 * 1024 * 1024:
        raise RuntimeError('Less than 16 GiB available host memory; no launch')
    disk_floors(out)
    for node, pci in DEVICES.items():
        if (Path('/sys/class/drm') / node / 'device').resolve().name != pci:
            raise RuntimeError('Render-node mapping changed; aborting')
    with ExitStack() as stack:
        locks = [stack.enter_context(open(p, 'a')) for p in LOCKS]
        for lock in locks:
            fcntl.flock(lock, fcntl.LOCK_EX | fcntl.LOCK_NB)
        out.mkdir(parents=True, exist_ok=False)
        dump(out / 'identity.json', identity)
        session = Session(out, tuple(lock.fileno() for lock in locks))
        def interrupt(signum, frame):
            session.interrupted = True
            session.interrupt_count += 1
        for sig in (signal.SIGTERM, signal.SIGINT):
            signal.signal(sig, interrupt)
        session.execute()


if __name__ == '__main__':
    main()
